#include<bits/stdc++.h>
using namespace std;

void solve(){
    long long a, b, n;
    cin>>a>>b>>n;
    
    long long arr[n];
    for(int i=0; i<n; i++) cin>>arr[i];
    
    long long ans=b;
    
    for(int i=0; i<n; i++){
        ans+= min(arr[i], a-1);
    }
    
    cout<<ans<<'\n';
    return;
}

int main(){
    int t;
    cin>>t;
    
    while(t--){
        solve();
    }
    
    return 0;
}
