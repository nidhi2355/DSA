#include<bits/stdc++.h>
using namespace std;

void solve(){
    int n, k;
    cin>>n>>k;
    
    vector<int> arr(n);
    for(int i=0; i<n; i++) cin>>arr[i];
    
    int ans= INT_MAX;
    int cnt=0;
    for(int i=0; i<n; i++){
        int rem= arr[i]%k;
        if(arr[i]%2 == 0) cnt++;
        if(rem==0){
            cout<<0<<endl;
            return;
        }
        ans= min(ans, k-rem);
    }
    
    if(k==4){
        if(cnt>=2) ans= min(ans, 0);
        else if(cnt>=1) ans= min(ans, 1);
        else ans= min(ans, 2);
    }
    
    cout<<ans<<endl;
    return;
}

int main(){
    int t;
    cin>>t;
    
    while(t--){
        solve();
    }
    
    return 0;
}
