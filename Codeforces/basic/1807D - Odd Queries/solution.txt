#include <bits/stdc++.h>
using namespace std;

void solve(){
    int n, q;
    cin >> n >> q;
    
    vector<int> arr(n);
    for(int i = 0; i < n; i++) cin >> arr[i];
    
    vector<int> odd(n + 1, 0);
    for(int i = 0; i < n; i++){
        odd[i + 1] = odd[i] + (arr[i] % 2);
    }
    
    int totalodd = odd[n];
    
    while(q--){
        int l, r, k;
        cin >> l >> r >> k;
        
        // Sum of odds outside the range [l, r]
        int odds_in_range = odd[r] - odd[l - 1];
        int outside_odds = totalodd - odds_in_range;
        
        // Odd count contributed by replacing range with k
        int replaced_odds = (k % 2 != 0) ? (r - l + 1) : 0;
        
        int final_odd_count = outside_odds + replaced_odds;
        
        if(final_odd_count % 2 != 0){
            cout << "YES\n";
        } else {
            cout << "NO\n";
        }
    }
}

int main(){
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    
    int t;
    cin >> t;
    while(t--){
        solve();
    }
    
    return 0;
}
