#include<bits/stdc++.h>
using namespace std;

int main(){
    int t;
    cin>>t;
    
    while(t--){
        int n;
        cin>>n;
        
        vector<int> arr(n);
        for(int i=0; i<n; i++) cin>>arr[i];
        
        int cnt=0, ind=0;
        
        while(ind<n){
            if(arr[ind]==0){
                ind++;
                continue;
            }
            int right= ind+1;
            while(right<n and arr[right]!= 0){
                right++;
            }
            ind= right+1;
            cnt++;
        }
        
        cout<<min(cnt, 2)<<'\n';
    }
    
    return 0;
}
