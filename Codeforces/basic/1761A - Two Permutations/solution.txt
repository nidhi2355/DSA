#include<bits/stdc++.h>
using namespace std;

int main(){
    int t;
    cin>>t;
    
    while(t--){
        int n, a, b;
        cin>>n>>a>>b;
        
        if((a==b and a==n) or (a+b+2<=n)){
            cout<<"YES\n";
        }
        else{
            cout<<"NO\n";
        }
    }
    
    return 0;
}
