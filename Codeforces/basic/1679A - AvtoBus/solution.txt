#include<bits/stdc++.h>
using namespace std;

long long cntmin(long long n){
    long long ans= n/6;
    
    if(n%6 ==2 or n%6== 4){
        ans++;
    }
    
    return ans;
}

long long cntmax(long long n){
    long long ans= n/4;
    
    return ans;
}

int main(){
    int t;
    cin>>t;
    
    while(t--){
        long long n;
        cin>>n;
        
        if(n%2 != 0 or n<4) {
            cout<<-1<<'\n';
            continue;
        }
        
        long long minans= cntmin(n);
        long long maxans= cntmax(n);
        
        cout<<minans<<' '<<maxans<<'\n';
    }
    
    return 0;
}
