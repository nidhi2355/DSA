#include<bits/stdc++.h>
using namespace std;

int main(){
    int t;
    cin>>t;
    
    while(t--){
        int n;
        cin>>n;
        
        vector<int> arr(n);
        for(int i=0; i<n; i++) cin>>arr[i];
        
        if(n==1){
            cout<<0<<'\n';
            continue;
        }
        
        int cnt=0;
        
        int right= 1;
        
        for(int i=0; i<n-1; ){
            if((arr[i]%2==0 and arr[right]%2!=0)
            or (arr[i]%2!=0 and arr[right]%2==0)){
                right++;
                i++;
            }
            else{
                arr[right]= arr[i]*arr[right];
                cnt++;
                i= right;
                right++;
            }
        }
        
        cout<<cnt<<'\n';
    }
    
    return 0;
}
