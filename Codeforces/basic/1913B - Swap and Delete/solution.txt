#include<bits/stdc++.h>
using namespace std;

void solve(){
    string s;
    cin>>s;
    
    int n= s.size();
    
    string t= "";
    int cnt1=0, cnt0=0;
    
    for(int i=0; i<n; i++){
        if(s[i]=='1') {
            cnt1++;
            t+='0';
        }
        else{
            cnt0++;
            t+='1';
        }
    }
    
    int ind=0;
    while(ind<n){
        if(t[ind]=='1'){
            if(cnt1){
                cnt1--;
            }
            else break;
        }
        else{
            if(cnt0){
                cnt0--;
            }
            else break;
        }
        ind++;
    }
    
    cout<<n-ind<<'\n';
    return;
}

int main(){
    int t;
    cin>>t;
    
    while(t--){
        solve();
    }
    
    return 0;
}
