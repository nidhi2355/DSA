#include<bits/stdc++.h>
using namespace std;

int main(){
    int t;
    cin>>t;
    
    while(t--){
        int n;
        cin>>n;
        
        vector<int> arr(n);
        
        for(int i=0; i<n; i++) cin>>arr[i];
        
        int cnt=0;
        
        vector<int> pref(n);
        
        for(int i=0; i<n; i++){
            if(arr[i]==2){
                cnt++;
            }
            
            pref[i]= cnt;
        }
        
        if(cnt%2==1){
            cout<<-1<<'\n';
            continue;
        }
        
        else{
            cnt= cnt/2;
            
            for(int i=0; i<n; i++){
                if(pref[i]== cnt){
                    cout<<i+1<<'\n';
                    break;
                }
            }
        }
    }
    
    return 0;
}
