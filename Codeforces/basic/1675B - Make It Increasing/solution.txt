#include<bits/stdc++.h>
using namespace std;

int main(){
    int t;
    cin>>t;
    
    while(t--){
        int n;
        cin>>n;
        
        vector<int> arr(n);
        for(int i=0; i<n; i++) cin>>arr[i];
        
        int op=0, flag=0;
        
        for(int i= n-2; i>=0; i--){
            if(arr[i+1]==0){
                flag=1;
                break;
            }
            
            while(arr[i]>0 and arr[i]>= arr[i+1]){
                arr[i]/=2;
                op++;
            }
        }
        
        if(flag==1){
            cout<<-1<<'\n';
        }
        else{
            cout<<op<<'\n';
        }
    }
    
    return 0;
}
