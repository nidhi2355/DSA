#include <bits/stdc++.h>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int t;
    cin >> t;

    while (t--) {
        long long n, k, x;
        cin >> n >> k >> x;

        long long minsum = k * (k + 1) / 2;
        long long maxsum = (n * (n + 1) / 2) - ((n - k) * (n - k + 1) / 2);

        if (x >= minsum && x <= maxsum) {
            cout << "YES\n";
        } else {
            cout << "NO\n";
        }
    }

    return 0;
}
