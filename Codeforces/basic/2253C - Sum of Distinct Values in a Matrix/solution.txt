#include<bits/stdc++.h>
using namespace std;

void solve(){
    int n, m, x, y;
    cin>>n>>m>>x>>y;
    
    vector<int> a(x);
    vector<int> b(y);
    
    for(int i=0; i<x; i++) cin>>a[i];
    
    for(int i=0; i<y; i++) cin>>b[i];
    
    int i= x-1, j= y-1;
    
    int c_a=0, c_b=0, c_both=0;
    
    long long maxcost=0;
    int maxcap= m+n-1;
    
    while((i>=0 || j>=0) && (c_a+ c_b+ c_both< maxcap)){
        if(i>=0 && (j<0 || a[i]>b[j])){
            if(c_a<n){
                c_a++;
                maxcost+= a[i];
            }
            i--;
        }
        else if(j>=0 && (i<0 || b[j]>a[i])){
            if(c_b<m){
                c_b++;
                maxcost+= b[j];
            }
            j--;
        }
        else{
            c_both++;
            maxcost+= a[i];
            i--;
            j--;
        }
    }
    
    cout<<maxcost<<'\n';
}

int main(){
    int t;
    cin>>t;
    
    while(t--){
        solve();
    }
}
